סימולציית דייקסטרה (לפי הפסאודו-קוד הרשמי)

צעד 0/0
// Step 1:
for each vertex v ∈ V // מעבר על כל צומת בגרף
d[v] = ∞ and π[v] = NIL // אתחול מרחק לאינסוף ואב לריק
d[s] = 0 // מרחק המקור (s) לעצמו הוא 0
S ← ∅ ; Q ← V ; // S ריקה, תור Q מכיל את כל הצמתים
// Step 2:
while Q ≠ ∅ // כל עוד התור לא ריק
do u ← EXTRACT-MIN(Q) // שולפים קדקוד עם מרחק מינימלי
if d[u] = ∞ then break // הגענו לקדקודים מנותקים - עוצרים
S ← S ∪ {u} // הוספת u לקבוצת המטופלים
for each vertex v ∈ Adj[u] ∩ Q // מעבר על שכני u שעדיין בתור Q
if d[u] + w(u,v) < d[v]: // ביצוע הקלה (RELAX)
d[v] = d[u] + w(u,v) // עדכון מרחק חדש וקצר יותר
π[v] = u // שמירת u בתור האבא של v
Update Q (DECREASE_KEY) // עדכון המיקום החדש בתור עדיפויות
return {d[v], π[v] : v ∈ V} // סיום והחזרת המערכים
מקרא סימנים:
  • d - המרחק המינימלי מצומת המקור.
  • π - אב (Pi): הצומת הקודם במסלול.
  • S - קבוצת הצמתים שטופלו סופית.
  • w - משקל הצלע.
